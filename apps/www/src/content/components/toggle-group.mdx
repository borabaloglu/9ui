## Usage

```tsx title="Imports"
import {
	ToggleGroup,
	ToggleGroupItem,
	ToggleGroupSeparator,
} from "@/components/ui/toggle-group"
```

```tsx title="Anatomy"
<ToggleGroup>
	<ToggleGroupItem />
	<ToggleGroupSeparator />
	<ToggleGroupItem />
</ToggleGroup>
```

## Tips

<Tips>

<Tip title="Name the group">

Give the group an `aria-label`.

```tsx {1}
<ToggleGroup aria-label="Text alignment">
	<ToggleGroupItem value="start">Start</ToggleGroupItem>
	<ToggleGroupItem value="end">End</ToggleGroupItem>
</ToggleGroup>
```

</Tip>

<Tip title="Label icon-only items">

Give each icon-only `ToggleGroupItem` its own `aria-label`.

```tsx {2,5}
<ToggleGroup aria-label="Text alignment">
	<ToggleGroupItem value="start" aria-label="Align start">
		<TextAlignStartIcon />
	</ToggleGroupItem>
	<ToggleGroupItem value="end" aria-label="Align end">
		<TextAlignEndIcon />
	</ToggleGroupItem>
</ToggleGroup>
```

</Tip>

<Tip title="Give each item a value">

Give each `ToggleGroupItem` a `value`.

```tsx {2-3}
<ToggleGroup defaultValue={["start"]} aria-label="Text alignment">
	<ToggleGroupItem value="start">Start</ToggleGroupItem>
	<ToggleGroupItem value="end">End</ToggleGroupItem>
</ToggleGroup>
```

</Tip>

</Tips>
