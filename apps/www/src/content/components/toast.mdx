<Step>Create the `use-toast` hook.</Step>

```tsx title="hooks/use-toast.tsx"
"use client";

import { Toast } from "@base-ui/react/toast";

import type { ButtonProps } from "@/components/ui/button";

type BaseToastManager = ReturnType<typeof Toast.createToastManager>;

type WithActionProps<Options> = Omit<Options, "actionProps"> & {
	actionProps?: Omit<ButtonProps, "size">;
};

type ToastManager = Omit<BaseToastManager, "add" | "update"> & {
	add: (
		options: WithActionProps<Parameters<BaseToastManager["add"]>[0]>,
	) => string;
	update: (
		toastId: string,
		options: WithActionProps<Parameters<BaseToastManager["update"]>[1]>,
	) => void;
};

export const toastManager = Toast.createToastManager() as ToastManager;

export const anchoredToastManager = Toast.createToastManager() as ToastManager;

export function useToast() {
	return Toast.useToastManager() as Omit<
		ReturnType<typeof Toast.useToastManager>,
		"add" | "update"
	> &
		Pick<ToastManager, "add" | "update">;
}
```

## Usage

```tsx title="Imports"
import { useToast } from "@/hooks/use-toast"
```

```tsx title="Toast"
const toast = useToast()

toast.add({
	title: "Event added to calendar",
	description: "Wednesday, April 23 at 4:20 PM",
})
```

To show a toast outside a component, use `toastManager`.

```tsx title="Outside React"
import { toastManager } from "@/hooks/use-toast"

toastManager.add({ title: "Event added to calendar" })
```

To move the toasts, set `position` on `ToastProvider`.

<ComponentPreview name="toast-position-demo" />

### Anchored

To attach a toast to an element, wrap it in `AnchoredToastProvider` and set `positionerProps.anchor`.

<ComponentPreview name="toast-anchored-demo" />

### RTL

To show the toast right to left, set `dir="rtl"` on `html` and wrap your app in `DirectionProvider`. See [RTL](/docs/getting-started/rtl). `position` is the screen corner in both directions, so to show toasts at the bottom start, set `position="bottom-right"`.
