## Usage

```tsx title="Imports"
import {
  InputGroup,
  InputGroupAddon,
  InputGroupButton,
  InputGroupInput,
  InputGroupText,
  InputGroupTextarea,
} from "@/components/ui/input-group"
```

```tsx title="Anatomy"
<InputGroup>
  <InputGroupAddon align="inline-start">
    <InputGroupText />
  </InputGroupAddon>
  <InputGroupInput />
  <InputGroupAddon align="inline-end">
    <InputGroupButton />
  </InputGroupAddon>
</InputGroup>
```

Give the control a `<label>` or `aria-label`, and give icon-only buttons an accessible name.
