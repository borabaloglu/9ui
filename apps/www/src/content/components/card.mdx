## Usage

```tsx title="Imports"
import {
	Card,
	CardAction,
	CardContent,
	CardDescription,
	CardFooter,
	CardHeader,
	CardTitle,
} from "@/components/ui/card"
```

```tsx title="Anatomy"
<Card>
	<CardHeader>
		<CardTitle />
		<CardDescription />
		<CardAction />
	</CardHeader>
	<CardContent />
	<CardFooter />
</Card>
```

## Tips

<Tips>

<Tip title="Make the title a heading">

To make the title a heading, set `render` on `CardTitle`.

```tsx {3}
<Card>
	<CardHeader>
		<CardTitle render={<h2 />}>Create project</CardTitle>
		<CardDescription>Deploy your new project in one click.</CardDescription>
	</CardHeader>
</Card>
```

</Tip>

</Tips>
