<Step>Create the `use-mobile` hook.</Step>

```tsx title="hooks/use-mobile.ts"
import * as React from "react";

const MOBILE_QUERY = "(max-width: 767px)";

function subscribe(callback: () => void) {
	const query = window.matchMedia(MOBILE_QUERY);
	query.addEventListener("change", callback);
	return () => query.removeEventListener("change", callback);
}

function getSnapshot() {
	return window.matchMedia(MOBILE_QUERY).matches;
}

function getServerSnapshot() {
	return false;
}

export function useIsMobile() {
	return React.useSyncExternalStore(subscribe, getSnapshot, getServerSnapshot);
}
```

## Usage

```tsx title="Imports"
import {
	Sidebar,
	SidebarContent,
	SidebarFooter,
	SidebarGroup,
	SidebarGroupAction,
	SidebarGroupContent,
	SidebarGroupLabel,
	SidebarHeader,
	SidebarInset,
	SidebarMenu,
	SidebarMenuAction,
	SidebarMenuBadge,
	SidebarMenuButton,
	SidebarMenuItem,
	SidebarMenuSkeleton,
	SidebarMenuSub,
	SidebarMenuSubButton,
	SidebarMenuSubItem,
	SidebarProvider,
	SidebarRail,
	SidebarSeparator,
	SidebarTrigger,
	useSidebar,
} from "@/components/ui/sidebar"
```

```tsx title="Anatomy"
<SidebarProvider>
	<Sidebar>
		<SidebarHeader />
		<SidebarContent>
			<SidebarGroup>
				<SidebarGroupLabel />
				<SidebarGroupAction />
				<SidebarGroupContent>
					<SidebarMenu>
						<SidebarMenuItem>
							<SidebarMenuButton />
							<SidebarMenuAction />
							<SidebarMenuBadge />
							<SidebarMenuSub>
								<SidebarMenuSubItem>
									<SidebarMenuSubButton />
								</SidebarMenuSubItem>
							</SidebarMenuSub>
						</SidebarMenuItem>
					</SidebarMenu>
				</SidebarGroupContent>
			</SidebarGroup>
		</SidebarContent>
		<SidebarFooter />
		<SidebarRail />
	</Sidebar>
	<SidebarInset>
		<SidebarTrigger />
	</SidebarInset>
</SidebarProvider>
```

To toggle the sidebar, press `⌘B` or `Ctrl+B`.

### Collapsing

Set `collapsible` on `Sidebar` to choose how it collapses.

| Value       | Behavior                              |
| ----------- | ------------------------------------- |
| `offcanvas` | Slides out of view. This is the default. |
| `icon`      | Shrinks to a strip of icons.          |
| `none`      | Never collapses.                      |

To show a label next to a button while the sidebar is collapsed to icons, set `tooltip` on `SidebarMenuButton`.

To hide a part while the sidebar is collapsed to icons, set `className="group-data-[collapsible=icon]/sidebar:hidden"` on it.

### Width

To change the width, set `--sidebar-width` and `--sidebar-width-icon` on `SidebarProvider`.

```tsx
<SidebarProvider
	style={
		{
			"--sidebar-width": "20rem",
			"--sidebar-width-icon": "3.5rem",
		} as React.CSSProperties
	}
/>
```

### Remembering the State

The sidebar saves its state in a `sidebar_state` cookie. To restore it on the server, read the cookie and pass it to `defaultOpen`.

```tsx title="app/layout.tsx"
import { cookies } from "next/headers"

export default async function Layout({
	children,
}: {
	children: React.ReactNode
}) {
	const cookieStore = await cookies()
	const defaultOpen = cookieStore.get("sidebar_state")?.value !== "false"

	return (
		<SidebarProvider defaultOpen={defaultOpen}>
			<AppSidebar />
			<SidebarInset>{children}</SidebarInset>
		</SidebarProvider>
	)
}
```

### Reading the State

To read or change the sidebar from your own components, call `useSidebar` inside `SidebarProvider`.

```tsx
const { state, open, setOpen, openMobile, setOpenMobile, isMobile, toggleSidebar } =
	useSidebar()
```

## Tips

<Tips>

<Tip title="Set up the layout">

Wrap your layout in `SidebarProvider` and put the page content in `SidebarInset`.

```tsx {1,5}
<SidebarProvider>
	<Sidebar>
		<SidebarContent />
	</Sidebar>
	<SidebarInset>{children}</SidebarInset>
</SidebarProvider>
```

</Tip>

<Tip title="Render a menu button as a link">

To render a menu button as a link, set `render` on `SidebarMenuButton`.

```tsx {1}
<SidebarMenuButton render={<Link href="/inbox" />}>
	<InboxIcon />
	<span>Inbox</span>
</SidebarMenuButton>
```

</Tip>

<Tip title="Mark the current page">

To mark the current page, set `active` on `SidebarMenuButton`.

```tsx {1}
<SidebarMenuButton active render={<Link href="/inbox" />}>
	<InboxIcon />
	<span>Inbox</span>
</SidebarMenuButton>
```

</Tip>

</Tips>

### SidebarTrigger

A ghost icon `Button` that toggles the sidebar. Takes Button props and shows a panel icon when it has no children.

### SidebarRail

A thin `button` along the sidebar edge that toggles the sidebar.

### SidebarInset

The page content next to the sidebar, a `main`.

### SidebarHeader

A `div` at the top of the sidebar.

### SidebarFooter

A `div` at the bottom of the sidebar.

### SidebarSeparator

A `Separator` inset to the sidebar padding.

### SidebarContent

The scrolling area between the header and the footer, a `div`.

### SidebarGroup

A `div` that holds a label, an action and a menu.

### SidebarGroupLabel

A `div` that fades out when the sidebar collapses to icons. To change the element, set `render`.

### SidebarGroupAction

A `button` at the end of the group label. To change the element, set `render`.

### SidebarGroupContent

A `div` that holds the group's content.

### SidebarMenu

A `ul` of menu items.

### SidebarMenuItem

An `li` that holds a menu button, an action, a badge and a submenu.
