## Usage

```tsx title="Imports"
import {
	Toolbar,
	ToolbarButton,
	ToolbarGroup,
	ToolbarInput,
	ToolbarLink,
	ToolbarSeparator,
} from "@/components/ui/toolbar"
```

```tsx title="Anatomy"
<Toolbar>
	<ToolbarButton />
	<ToolbarLink />
	<ToolbarSeparator />
	<ToolbarGroup>
		<ToolbarButton />
		<ToolbarButton />
	</ToolbarGroup>
	<ToolbarInput />
</Toolbar>
```

`ToolbarButton` and `ToolbarInput` have no styles of their own. Use the `render` prop to compose them with other components.
