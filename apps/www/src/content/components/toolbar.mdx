## Usage

```tsx title="Imports"
import {
	Toolbar,
	ToolbarButton,
	ToolbarGroup,
	ToolbarInput,
	ToolbarLink,
	ToolbarSeparator,
} from "@/components/ui/toolbar"
```

```tsx title="Anatomy"
<Toolbar>
	<ToolbarButton />
	<ToolbarLink />
	<ToolbarSeparator />
	<ToolbarGroup>
		<ToolbarButton />
		<ToolbarButton />
	</ToolbarGroup>
	<ToolbarInput />
</Toolbar>
```

## Tips

<Tips>

<Tip title="Label the toolbar">

Give the toolbar an `aria-label`.

```tsx {1}
<Toolbar aria-label="File actions">
	<ToolbarButton render={<Button variant="ghost" />}>Share</ToolbarButton>
</Toolbar>
```

</Tip>

<Tip title="Label icon-only buttons">

Give each icon-only button its own `aria-label`.

```tsx {3}
<ToolbarButton
	render={<Button variant="ghost" size="icon" />}
	aria-label="Download"
>
	<DownloadIcon />
</ToolbarButton>
```

</Tip>

<Tip title="Compose with other components">

Set `render` on `ToolbarButton` and `ToolbarInput` to compose them with other components.

```tsx {2-3}
<Toolbar aria-label="Table tools">
	<ToolbarButton render={<Button variant="ghost" />}>Filter</ToolbarButton>
	<ToolbarInput render={<Input placeholder="Search" aria-label="Search" />} />
</Toolbar>
```

</Tip>

</Tips>
