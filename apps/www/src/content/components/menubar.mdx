## Usage

```tsx title="Imports"
import {
	Menubar,
	MenubarCheckboxItem,
	MenubarGroup,
	MenubarGroupLabel,
	MenubarItem,
	MenubarLinkItem,
	MenubarMenu,
	MenubarPopup,
	MenubarRadioGroup,
	MenubarRadioItem,
	MenubarSeparator,
	MenubarShortcut,
	MenubarSub,
	MenubarSubPopup,
	MenubarSubTrigger,
	MenubarTrigger,
} from "@/components/ui/menubar"
```

```tsx title="Anatomy"
<Menubar>
	<MenubarMenu>
		<MenubarTrigger />
		<MenubarPopup>
			<MenubarGroup>
				<MenubarGroupLabel />
				<MenubarItem>
					<MenubarShortcut />
				</MenubarItem>
				<MenubarLinkItem />
			</MenubarGroup>
			<MenubarSeparator />
			<MenubarCheckboxItem />
			<MenubarRadioGroup>
				<MenubarRadioItem />
			</MenubarRadioGroup>
			<MenubarSub>
				<MenubarSubTrigger />
				<MenubarSubPopup>
					<MenubarItem />
				</MenubarSubPopup>
			</MenubarSub>
		</MenubarPopup>
	</MenubarMenu>
</Menubar>
```

The parts inside `MenubarPopup` work like the ones in [Menu](/docs/components/menu).
