## Usage

```tsx title="Imports"
import {
	Command,
	CommandCollection,
	CommandDialog,
	CommandEmpty,
	CommandFooter,
	CommandGroup,
	CommandGroupLabel,
	CommandInput,
	CommandItem,
	CommandList,
	CommandSeparator,
	CommandShortcut,
} from "@/components/ui/command"
```

```tsx title="Anatomy"
<Command>
	<CommandInput />
	<CommandEmpty />
	<CommandList>
		<CommandGroup>
			<CommandGroupLabel />
			<CommandCollection />
		</CommandGroup>
		<CommandSeparator />
		<CommandItem>
			<CommandShortcut />
		</CommandItem>
	</CommandList>
	<CommandFooter />
</Command>
```

## Tips

<Tips>

<Tip title="Label the input">

Give `CommandInput` an `aria-label`.

```tsx {3}
<InputGroup>
	<InputGroupInput
		aria-label="Search commands"
		render={<CommandInput />}
	/>
</InputGroup>
```

</Tip>

<Tip title="Use Autocomplete props">

`Command` takes the same props as [Autocomplete](/docs/components/autocomplete), like `items` and `filter`.

```tsx {1}
<Command items={pages} filter={null}>
	<CommandList>
		{(page: Page) => (
			<CommandItem key={page.value} value={page}>
				{page.label}
			</CommandItem>
		)}
	</CommandList>
</Command>
```

</Tip>

<Tip title="Style the input as an Input Group">

To style `CommandInput` as an [Input Group](/docs/components/input-group), set it as the `render` of `InputGroupInput`.

```tsx {5}
<InputGroup>
	<InputGroupInput
		placeholder="Type a command or search…"
		aria-label="Search commands"
		render={<CommandInput />}
	/>
</InputGroup>
```

</Tip>

<Tip title="Keep items clear of the scrollbar">

To keep items clear of the scrollbar, set `scrollbarGutter` on `CommandList`.

```tsx {1}
<CommandList scrollbarGutter>
	<CommandItem value="settings">Settings</CommandItem>
</CommandList>
```

</Tip>

<Tip title="Run a command">

To run a command, set `onClick` on `CommandItem`.

```tsx {1}
<CommandItem value="new-document" onClick={() => createDocument()}>
	New document
</CommandItem>
```

</Tip>

</Tips>
