## Usage

```tsx title="Imports"
import {
	EmojiPicker,
	EmojiPickerContent,
	EmojiPickerEmpty,
	EmojiPickerFooter,
	EmojiPickerList,
	EmojiPickerLoading,
	EmojiPickerPreview,
	EmojiPickerSearch,
	EmojiPickerSkinToneSelector,
} from "@/components/ui/emoji-picker"
```

```tsx title="Anatomy"
<EmojiPicker>
	<EmojiPickerSearch />
	<EmojiPickerContent>
		<EmojiPickerLoading />
		<EmojiPickerEmpty />
		<EmojiPickerList />
	</EmojiPickerContent>
	<EmojiPickerFooter>
		<EmojiPickerPreview />
		<EmojiPickerSkinToneSelector />
	</EmojiPickerFooter>
</EmojiPicker>
```

To get the picked emoji, set `onEmojiSelect` on `EmojiPicker`.
