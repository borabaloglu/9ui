## Anatomy

Put each control in a `Field` with a `name`, and add a `FieldError` for each message.

```tsx {1,4-11}
<Form onFormSubmit={(values) => console.log(values)}>
	<Field name="name">
		<FieldLabel>Name</FieldLabel>
		<Input required minLength={3} pattern="^[a-zA-Z]+$" />
		<FieldError match="valueMissing">Name is required.</FieldError>
		<FieldError match="tooShort">
			Name must be at least 3 characters.
		</FieldError>
		<FieldError match="patternMismatch">
			Name must only contain letters.
		</FieldError>
	</Field>

	<Button type="submit">Send</Button>
</Form>
```

- `Form` handles the submission and passes server errors to the fields.
- `Input` takes native constraints such as `required`, `minLength` and `pattern`.
- `FieldError` shows a message for a native constraint, your own validation or a server error.

## Form

### Submit Values as an Object

To get the field values as an object, use `onFormSubmit`.

```tsx {2-4}
<Form
	onFormSubmit={(values) => {
		// values: Record<string, unknown>
	}}
/>
```

### Submit with the Native Event

To read the submit event or `FormData`, use `onSubmit`.

```tsx {2-6}
<Form
	onSubmit={(event) => {
		event.preventDefault()
		const formData = new FormData(event.currentTarget)
	}}
/>
```

## Submitting

When a submit fails validation, `Form` moves focus to the first invalid field.

### Pending State

To show progress while the form submits, set `loading` on the submit button.

```tsx {1,5-10,13}
const [loading, setLoading] = useState(false)

<Form
	onFormSubmit={async (values) => {
		setLoading(true)
		try {
			await submit(values)
		} finally {
			setLoading(false)
		}
	}}
>
	<Button type="submit" loading={loading}>
		Send
	</Button>
</Form>
```

## Validation

### Client-Side Validation

Set native constraints on the control and add a `FieldError` with `match` for each message.

```tsx {2-7}
<Field name="username">
	<Input required minLength={3} pattern="^[a-zA-Z0-9_]+$" />
	<FieldError match="valueMissing">Username is required.</FieldError>
	<FieldError match="tooShort">At least 3 characters.</FieldError>
	<FieldError match="patternMismatch">
		Use letters, numbers or underscores.
	</FieldError>
</Field>
```

### Validation Modes

To choose when validation runs, set `validationMode` on `Form` or `Field`.

```tsx {1}
<Form validationMode="onBlur">{/* ... */}</Form>
```

| Mode | Description |
| --- | --- |
| `"onSubmit"` | Validates on submit, then on every change. This is the default. |
| `"onBlur"` | Validates when a field loses focus. |
| `"onChange"` | Validates on every change. |

### Custom Validation

To run your own check, set `validate` on `Field`. Use `FieldError` with `match="customError"` to show the message it returns.

<ComponentPreview name="form-base-ui-custom-validation-demo" />

```tsx {3-9,12}
<Field
	name="username"
	validate={(value) => {
		const username = String(value ?? "").trim()
		if (username === "admin") {
			return "Username cannot be 'admin'."
		}
		return null
	}}
>
	<Input />
	<FieldError match="customError" />
</Field>
```

To delay validation while typing, set `validationMode="onChange"` and `validationDebounceTime`.

```tsx {3-4}
<Field
	name="username"
	validationMode="onChange"
	validationDebounceTime={300}
	validate={checkUsername}
>
	<Input />
	<FieldError match="customError" />
</Field>
```

## Server-Side Errors

To show errors returned by the server, pass them to `errors` on `Form` and add a `FieldError` without `match`. Submit the form to see one.

<ComponentPreview name="form-base-ui-server-errors-demo" />

```tsx {1,4-8,12}
const [errors, setErrors] = useState({})

<Form
	errors={errors}
	onFormSubmit={async (values) => {
		const response = await submit(values)
		setErrors(response.errors ?? {})
	}}
>
	<Field name="username">
		<Input />
		<FieldError />
	</Field>
</Form>
```

### Server Actions

To submit to a server action, call it in `onFormSubmit` inside a transition and pass the errors it returns to `errors`.

```tsx {8-9,13-19} title="form.tsx"
"use client"

import { useState, useTransition } from "react"

import { createUser } from "./actions"

export default function ExampleForm() {
	const [errors, setErrors] = useState({})
	const [pending, startTransition] = useTransition()

	return (
		<Form
			errors={errors}
			onFormSubmit={(values) => {
				startTransition(async () => {
					const result = await createUser(values)
					setErrors(result.errors)
				})
			}}
		>
			<Field name="username">
				<FieldLabel>Username</FieldLabel>
				<Input required />
				<FieldError />
			</Field>
			<Button type="submit" loading={pending}>
				Save
			</Button>
		</Form>
	)
}
```

```ts title="actions.ts"
"use server"

export async function createUser(values: { username: string }) {
	if (values.username === "admin") {
		return { errors: { username: "'admin' is reserved for system use" } }
	}

	return { errors: {} }
}
```

## Using with Zod

To validate with a schema, parse the values in `onFormSubmit` and pass the field errors to `errors`.

<ComponentPreview name="form-base-ui-zod-demo" />

<Steps>

<Step>Install Zod.</Step>

```bash title="Terminal"
npm install zod
```

<Step>Parse the values and set the errors.</Step>

```tsx {4-7,10,13-17,19,23} title="form.tsx"
import type { Form as FormPrimitive } from "@base-ui/react/form"
import { z } from "zod"

const schema = z.object({
	name: z.string().min(1, "Name is required."),
	message: z.string().min(10, "Message must be at least 10 characters."),
})

export default function ExampleForm() {
	const [errors, setErrors] = useState<FormPrimitive.Props["errors"]>({})

	const handleSubmit = (values: FormPrimitive.Values) => {
		const result = schema.safeParse(values)
		if (!result.success) {
			setErrors(z.flattenError(result.error).fieldErrors)
			return
		}

		setErrors({})
	}

	return (
		<Form errors={errors} onFormSubmit={handleSubmit}>
			{/* ... */}
		</Form>
	)
}
```

</Steps>

### Phone Input

Keep the number in state, check it on submit with `isValidPhoneNumber` from `react-phone-number-input` and set `invalid` on `Field`.

<ComponentPreview name="form-base-ui-phone-input-demo" />

```tsx {8-11,14,23}
import { isValidPhoneNumber } from "react-phone-number-input"

const [phoneNumber, setPhoneNumber] = useState("")
const [invalid, setInvalid] = useState(false)

<Form
	onFormSubmit={() => {
		if (!isValidPhoneNumber(phoneNumber)) {
			setInvalid(true)
			return
		}
	}}
>
	<Field name="phoneNumber" invalid={invalid}>
		<FieldLabel>Phone number</FieldLabel>
		<PhoneInput
			value={phoneNumber}
			onChange={(value) => {
				setPhoneNumber(value)
				setInvalid(false)
			}}
		/>
		<FieldError match={invalid}>Enter a valid phone number.</FieldError>
	</Field>
</Form>
```

### Date Picker

Keep the date in state, check it on submit and set `invalid` on `Field` and `aria-invalid` on the trigger. Give `FieldError` an `id` and pass it to `aria-describedby` on the trigger so the error is announced.

<ComponentPreview name="form-base-ui-date-picker-demo" />

```tsx {9-12,15,35}
const id = useId()
const errorId = useId()
const [open, setOpen] = useState(false)
const [date, setDate] = useState<Date | null>(null)
const [invalid, setInvalid] = useState(false)

<Form
	onFormSubmit={() => {
		if (!date) {
			setInvalid(true)
			return
		}
	}}
>
	<Field name="startDate" invalid={invalid}>
		<FieldLabel htmlFor={id}>Start date</FieldLabel>
		<Popover open={open} onOpenChange={setOpen}>
			<PopoverTrigger
				render={<Button id={id} variant="outline" aria-describedby={errorId} aria-invalid={invalid || undefined} />}
			>
				{date ? format(date, "MMM d, yyyy") : "Select a date"}
			</PopoverTrigger>
			<PopoverPopup className="w-auto p-0" align="start">
				<Calendar
					mode="single"
					selected={date ?? undefined}
					onSelect={(next) => {
						setDate(next ?? null)
						setInvalid(false)
						setOpen(false)
					}}
				/>
			</PopoverPopup>
		</Popover>
		<FieldError id={errorId} match={invalid}>Start date is required.</FieldError>
	</Field>
</Form>
```

## Array Fields

To let people add and remove fields, keep a list of row ids in state and render a `Field` with its own `name` for each row.

<ComponentPreview name="form-base-ui-array-demo" />

```tsx {1-2,5-6,14,26}
const [rows, setRows] = useState([0])
const nextRow = useRef(1)

<Form onFormSubmit={(values) => console.log(values)}>
	{rows.map((row, index) => (
		<Field key={row} name={`email-${row}`}>
			<div className="flex gap-2">
				<Input type="email" aria-label={`Email ${index + 1}`} required />
				<Button
					type="button"
					variant="ghost"
					size="icon"
					aria-label={`Remove email ${index + 1}`}
					onClick={() => setRows((prev) => prev.filter((item) => item !== row))}
				>
					<XIcon />
				</Button>
			</div>
			<FieldError match="valueMissing">Email is required.</FieldError>
			<FieldError match="typeMismatch">Enter a valid email address.</FieldError>
		</Field>
	))}
	<Button
		type="button"
		variant="outline"
		onClick={() => setRows((prev) => [...prev, nextRow.current++])}
	>
		Add email
	</Button>
</Form>
```
