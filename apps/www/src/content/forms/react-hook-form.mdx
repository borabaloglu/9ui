## Anatomy

Wrap each `Field` in a `Controller` and pass the field state to it.

```tsx {5-12}
<Controller
	name="username"
	control={control}
	render={({ field, fieldState }) => (
		<Field name={field.name} invalid={fieldState.invalid}>
			<FieldLabel>Username</FieldLabel>
			<Input {...field} />
			<FieldDescription>This is your public display name.</FieldDescription>
			{fieldState.invalid && (
				<FieldError match>{fieldState.error?.message}</FieldError>
			)}
		</Field>
	)}
/>
```

- `Controller` connects the control to React Hook Form.
- `invalid` on `Field` takes `fieldState.invalid`.
- `match` on `FieldError` shows the message from React Hook Form.

## Form

### Create a Schema

Define the form values with Zod, or any validation library that has a resolver.

```tsx title="form.tsx"
import { z } from "zod"

const schema = z.object({
	name: z.string().min(1, "Name is required."),
	message: z
		.string()
		.min(10, "Message must be at least 10 characters.")
		.max(200, "Message must be 200 characters or less."),
})

type FormValues = z.infer<typeof schema>
```

### Set Up the Form

Pass the schema to `useForm` with `zodResolver`.

```tsx {7-13,20} title="form.tsx"
import { zodResolver } from "@hookform/resolvers/zod"
import { Controller, useForm } from "react-hook-form"

// ...

export default function ExampleForm() {
	const { control, handleSubmit } = useForm<FormValues>({
		resolver: zodResolver(schema),
		defaultValues: {
			name: "",
			message: "",
		},
	})

	function onSubmit(data: FormValues) {
		console.log(data)
	}

	return (
		<form onSubmit={handleSubmit(onSubmit)}>{/* ... */}</form>
	)
}
```

Then add a `Controller` with a `Field` for each value. The demo at the top of this page has the full code.

## Submitting

When a submit fails validation, React Hook Form moves focus to the first invalid field whose control receives `field.ref`.

To show progress while the form submits, pass `formState.isSubmitting` to `loading` on the submit button.

```tsx {3,9}
const {
	handleSubmit,
	formState: { isSubmitting },
} = useForm<FormValues>({
	resolver: zodResolver(schema),
})

<form onSubmit={handleSubmit(onSubmit)}>
	<Button type="submit" loading={isSubmitting}>
		Send
	</Button>
</form>
```

## Validation

### Validation Modes

To choose when validation runs, set `mode` on `useForm`.

```tsx {3}
const { control, handleSubmit } = useForm<FormValues>({
	resolver: zodResolver(schema),
	mode: "onChange",
})
```

| Mode | Description |
| --- | --- |
| `"onSubmit"` | Validates on submit, then on every change. This is the default. |
| `"onBlur"` | Validates when a field loses focus. |
| `"onChange"` | Validates on every change. |
| `"onTouched"` | Validates on the first blur, then on every change. |
| `"all"` | Validates on blur and on change. |

## Server-Side Errors

To show an error returned by the server, call `setError`. Submit the form to see one.

<ComponentPreview name="form-react-hook-form-server-errors-demo" />

```tsx {1,11-14}
const { control, handleSubmit, setError } = useForm<FormValues>({
	resolver: zodResolver(schema),
	defaultValues: {
		username: "",
	},
})

async function onSubmit(data: FormValues) {
	const res = await createUser(data)

	if (!res.success) {
		setError("username", { message: res.message })
		return
	}
}
```

### Date Picker

Pass `field.value` to `selected` and `field.onChange` to `onSelect` on `Calendar`, and set `aria-invalid` and `ref={field.ref}` on the trigger. Use `null` as the empty value in `defaultValues`. Give `FieldError` an `id` and pass it to `aria-describedby` on the trigger so the error is announced.

<ComponentPreview name="form-react-hook-form-date-picker-demo" />

```tsx {13,20-22}
const id = useId()
const errorId = useId()
const [open, setOpen] = useState(false)

<Controller
	name="startDate"
	control={control}
	render={({ field, fieldState }) => (
		<Field name={field.name} invalid={fieldState.invalid}>
			<FieldLabel htmlFor={id}>Start date</FieldLabel>
			<Popover open={open} onOpenChange={setOpen}>
				<PopoverTrigger
					render={<Button id={id} variant="outline" aria-describedby={errorId} aria-invalid={fieldState.invalid || undefined} ref={field.ref} onBlur={field.onBlur} />}
				>
					{field.value ? format(field.value, "MMM d, yyyy") : "Select a date"}
				</PopoverTrigger>
				<PopoverPopup className="w-auto p-0" align="start">
					<Calendar
						mode="single"
						selected={field.value ?? undefined}
						onSelect={(next) => {
							field.onChange(next ?? null)
							setOpen(false)
						}}
					/>
				</PopoverPopup>
			</Popover>
			{fieldState.invalid && (
				<FieldError id={errorId} match>{fieldState.error?.message}</FieldError>
			)}
		</Field>
	)}
/>
```

## Array Fields

To let people add and remove fields, use `useFieldArray`. Render a `Controller` for each item, keyed by its `id`, and call `append` and `remove` to change the list.

<ComponentPreview name="form-react-hook-form-array-demo" />

```tsx {3-6,9-12,23,38}
import { useFieldArray } from "react-hook-form"

const { fields, append, remove } = useFieldArray({
	control,
	name: "emails",
})

<form onSubmit={handleSubmit(onSubmit)}>
	{fields.map((item, index) => (
		<Controller
			key={item.id}
			name={`emails.${index}.address`}
			control={control}
			render={({ field, fieldState }) => (
				<Field name={field.name} invalid={fieldState.invalid}>
					<div className="flex gap-2">
						<Input {...field} aria-label={`Email ${index + 1}`} />
						<Button
							type="button"
							variant="ghost"
							size="icon"
							aria-label={`Remove email ${index + 1}`}
							onClick={() => remove(index)}
						>
							<XIcon />
						</Button>
					</div>
					{fieldState.invalid && (
						<FieldError match>{fieldState.error?.message}</FieldError>
					)}
				</Field>
			)}
		/>
	))}
	<Button
		type="button"
		variant="outline"
		onClick={() => append({ address: "" })}
	>
		Add email
	</Button>
</form>
```
